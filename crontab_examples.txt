# 台股 AI 自動分析系統 - Crontab 排程範例

# 1. 自動盤前分析 (週一至週五 08:30 執行)
30 8 * * 1-5 cd /path/to/taiwan-stock-ai-analyzer && python3 main.py pre-market >> logs/pre_market.log 2>&1

# 2. 自動盤中實時監控 (週一至週五 09:00 啟動)
0 9 * * 1-5 cd /path/to/taiwan-stock-ai-analyzer && python3 main.py realtime >> logs/realtime.log 2>&1

# 3. 自動盤後體檢與 Telegram 推播 (週一至週五 15:00 執行)
0 15 * * 1-5 cd /path/to/taiwan-stock-ai-analyzer && python3 main.py post-market >> logs/post_market.log 2>&1

# 4. 夜間模型自動再訓練 (每週一至週五 22:00)
0 22 * * 1-5 cd /path/to/taiwan-stock-ai-analyzer && python3 main.py train --all >> logs/retraining.log 2>&1

# 5. 每週清理超過 14 天的舊日誌
0 2 * * 0 find /path/to/taiwan-stock-ai-analyzer/logs/ -name "*.log" -mtime +14 -delete
